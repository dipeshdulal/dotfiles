# Japanese + Nepali input: Fcitx5 framework, Mozc (Google Japanese Input) as
# the romaji -> kana/kanji engine, and m17n for Nepali. All official-repo
# packages -- no AUR. The Hyprland side (env vars and autostart) lives in
# ../hypr/hyprland.lua, not here.
fcitx5
fcitx5-mozc

# Nepali. m17n-db carries the MPP "ne-rom" romanized layout; fcitx5-m17n
# exposes it to fcitx5 as the input method m17n_ne_rom.
fcitx5-m17n
m17n-db

# GTK/Qt IM modules. Wayland-native GTK4/Qt6 apps use text-input-v3 and do
# not strictly need these, but XWayland apps do -- QT_IM_MODULE and
# XMODIFIERS are set in ../hypr/hyprland.lua, so the modules must exist.
fcitx5-gtk
fcitx5-qt

# GUI for tweaking groups/addons after setup.
fcitx5-configtool
